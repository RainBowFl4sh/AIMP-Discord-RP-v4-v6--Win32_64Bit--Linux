cmake_minimum_required(VERSION 3.20)
project(aimp_discord_rpc LANGUAGES CXX)

# Targets:
#   Windows x64 / x86 (Win32)  -> aimp_discord_rpc.dll   (MSVC: cmake -A x64 | -A Win32, or MinGW, see cmake/*.cmake)
#   Linux x86_64               -> aimp_discord_rpc.so    (native AIMP for Linux; build needs libcurl + cairo headers)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
    set(CMAKE_BUILD_TYPE Release)
endif()
# static runtime -> no VC++ redistributable needed on the user's PC
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")

# Folder that contains apiPlugin.h, apiCore.h, ... (AIMP SDK, C++ headers)
set(AIMP_SDK_DIR "${CMAKE_CURRENT_SOURCE_DIR}/sdk" CACHE PATH "Path to the AIMP SDK C++ headers")
if(NOT EXISTS "${AIMP_SDK_DIR}/apiPlugin.h")
    message(FATAL_ERROR "Set -DAIMP_SDK_DIR=<folder containing apiPlugin.h>")
endif()

# Translations (langs/*.lng) and the changelogs (CHANGELOG.md, langs/changelog.*.md) are compiled into the plugin: generated/embedded.h
set(EMBED_DIR "${CMAKE_CURRENT_BINARY_DIR}/generated")
set(EMBED_TEXT "// Generated by CMake from langs/*.lng and CHANGELOG.md - do not edit.\n#pragma once\n")
string(REPEAT "0x[0-9a-f][0-9a-f]," 24 EMBED_ROW)
foreach(entry "kLangEn=langs/english.lng" "kLangDe=langs/german.lng" "kLangRu=langs/russian.lng"
              "kLangUk=langs/ukrainian.lng" "kChangelog=CHANGELOG.md" "kChangelogDe=langs/changelog.de.md"
              "kChangelogRu=langs/changelog.ru.md" "kChangelogUk=langs/changelog.uk.md")
    string(REPLACE "=" ";" entry "${entry}")
    list(GET entry 0 var)
    list(GET entry 1 file)
    file(READ "${CMAKE_CURRENT_SOURCE_DIR}/${file}" hex HEX)
    string(REGEX REPLACE "([0-9a-f][0-9a-f])" "0x\\1," hex "${hex}")
    string(REGEX REPLACE "(${EMBED_ROW})" "\\1\n" hex "${hex}")
    string(APPEND EMBED_TEXT "static const unsigned char ${var}[] = {\n${hex}0};\n")
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/${file}")
endforeach()
file(WRITE "${EMBED_DIR}/embedded.h.new" "${EMBED_TEXT}")
configure_file("${EMBED_DIR}/embedded.h.new" "${EMBED_DIR}/embedded.h" COPYONLY)   # touched only when changed

add_library(aimp_discord_rpc SHARED
    src/plugin.cpp
    src/settings_ui.cpp
    src/preview.cpp
    src/presence.cpp
    src/discord_ipc.cpp
    src/cover.cpp
    src/jobs.cpp
    src/update.cpp
    src/web.cpp
    src/config.cpp
    src/i18n.cpp
    src/util.cpp)

# some SDK versions ship helper code in apiTypes.cpp - compile it if present
if(EXISTS "${AIMP_SDK_DIR}/apiTypes.cpp")
    target_sources(aimp_discord_rpc PRIVATE "${AIMP_SDK_DIR}/apiTypes.cpp")
endif()

target_include_directories(aimp_discord_rpc PRIVATE "${AIMP_SDK_DIR}" "${AIMP_SDK_DIR}/Helpers" src "${EMBED_DIR}")
set_target_properties(aimp_discord_rpc PROPERTIES PREFIX "" OUTPUT_NAME "aimp_discord_rpc")

# small binary: optimize for size, drop unused code (the plugin is I/O bound, speed is not the issue)
if(MSVC)
    target_compile_options(aimp_discord_rpc PRIVATE /utf-8 /GR- $<$<CONFIG:Release>:/O1 /Gy /GL>)
    target_link_options(aimp_discord_rpc PRIVATE $<$<CONFIG:Release>:/OPT:REF /OPT:ICF /LTCG>)
else()
    # the plugin neither throws nor catches nor uses RTTI: no exception tables / type info (about 7 % smaller)
    target_compile_options(aimp_discord_rpc PRIVATE -fno-exceptions -fno-rtti
        $<$<CONFIG:Release>:-Os -ffunction-sections -fdata-sections>)
    target_link_options(aimp_discord_rpc PRIVATE $<$<CONFIG:Release>:-Wl,--gc-sections -s>)
endif()

if(WIN32)
    target_sources(aimp_discord_rpc PRIVATE
        src/aimp_discord_rpc.def)   # undecorated export name, also for 32-bit (stdcall)
    target_compile_definitions(aimp_discord_rpc PRIVATE UNICODE _UNICODE NOMINMAX _CRT_SECURE_NO_WARNINGS)
    target_link_libraries(aimp_discord_rpc PRIVATE
        winhttp gdiplus shlwapi ole32 user32 gdi32 comdlg32)
    if(MINGW)
        # the SDK maps _ctoll to the non-existing wtoll for non-MSVC compilers
        target_compile_definitions(aimp_discord_rpc PRIVATE wtoll=_wtoll)
        target_link_options(aimp_discord_rpc PRIVATE -static -static-libgcc -static-libstdc++ -Wl,--enable-stdcall-fixup)
    endif()
else()
    # libcurl is loaded at run time (dlopen); cairo draws the preview (AIMP for Linux itself is built on it)
    find_path(CURL_HEADER_DIR curl/curl.h REQUIRED)
    find_package(PkgConfig REQUIRED)
    pkg_check_modules(CAIRO REQUIRED cairo)
    target_include_directories(aimp_discord_rpc PRIVATE ${CAIRO_INCLUDE_DIRS} ${CURL_HEADER_DIR})
    target_link_libraries(aimp_discord_rpc PRIVATE ${CAIRO_LIBRARIES} dl pthread)
    target_compile_options(aimp_discord_rpc PRIVATE -fvisibility=hidden -Wno-attributes)
    # libstdc++ is linked statically so the plugin does not depend on the distribution's compiler version;
    # only AIMPPluginGetHeader is exported (not the C++ library). Released builds: tools/build_linux_portable.sh
    target_link_options(aimp_discord_rpc PRIVATE -static-libstdc++ -static-libgcc -Wl,--no-undefined
        "-Wl,--version-script=${CMAKE_CURRENT_SOURCE_DIR}/src/exports.map")
endif()

# Mock AIMP host for tests/run_linux_test.sh and tests/run_wine_test.sh (not part of the plugin, not built by default)
add_executable(host_test EXCLUDE_FROM_ALL tests/host_test.cpp "${AIMP_SDK_DIR}/apiTypes.cpp")
target_include_directories(host_test PRIVATE "${AIMP_SDK_DIR}")
if(WIN32)
    target_compile_definitions(host_test PRIVATE UNICODE _UNICODE NOMINMAX)
    if(MINGW)
        target_compile_definitions(host_test PRIVATE wtoll=_wtoll)
        target_link_options(host_test PRIVATE -static -static-libgcc -static-libstdc++)
    endif()
    target_link_libraries(host_test PRIVATE ole32 oleaut32 user32 gdi32)
else()
    target_include_directories(host_test PRIVATE ${CAIRO_INCLUDE_DIRS})
    target_compile_options(host_test PRIVATE -Wno-attributes)
    target_link_libraries(host_test PRIVATE ${CAIRO_LIBRARIES} dl)
endif()
